# Enable sound effects on death
deathstats_enable_sounds (Enable Death Sounds) bool true

# Orbit camera around player corpse on death
deathstats_enable_camera (Cinematic Camera Orbit on Death) bool true

# Orbit camera radius in nodes
deathstats_orbit_radius (Deathcam Orbit Radius) float 3.2 1.0 10.0

# Orbit camera height above corpse in nodes
deathstats_orbit_height (Deathcam Orbit Height) float 1.5 0.2 6.0

# Orbit camera rotation speed in radians per second (~0.4 rad/s is ~15.7s per turn)
deathstats_orbit_speed (Deathcam Orbit Rotation Speed) float 0.4 0.1 3.0


# Enable zoom and fade animation for YOU DIED banner
deathstats_enable_animation (Enable Zoom & Fade Animation) bool true

# Duration of zoom & fade animation in seconds
deathstats_animation_duration (Animation Duration) float 2.4 0.2 5.0

# Opacity of blood splatter vignette overlay (0-255)
deathstats_blood_opacity (Blood Splatter Opacity) int 240 0 255

# Placement side for death statistics card (right, left, center)
deathstats_formspec_side (Statistics Formspec Side) enum right right,left,center

# Enable broken/fractured limb rotations on death (arms, legs, head)
deathstats_enable_limb_fractures (Enable Corpse Limb Fractures) bool true


# Enable particle effects on corpse (bubbles in water, fire in lava, smoke in fire, impact debris on ground)
deathstats_enable_corpse_particles (Enable Corpse Particles) bool true

# Enable the live scoreboard feature (hold-to-view HUD overlay and /scores table)
deathstats_enable_scoreboard (Enable Scoreboard Feature) bool true

# Key or key combination used to hold and display the live scoreboard HUD
# Options: zoom, sneak_aux1, aux1, sneak
deathstats_scoreboard_key (Scoreboard Activation Key) enum zoom zoom,sneak_aux1,aux1,sneak

# In-game time display format on scoreboard header (24h or 12h)
deathstats_time_format (Scoreboard Time Format) enum 24h 24h,12h

# Live scoreboard HUD update interval in seconds while key is held down
deathstats_scoreboard_update_interval (Scoreboard Live Update Rate) float 1.0 0.2 5.0

# Suppress on-screen chat while the scoreboard HUD overlay is held open
deathstats_scoreboard_suppress_chat (Suppress Chat on Scoreboard) bool true

# Enable mock players in scoreboard for testing large rosters and scrollable layouts
deathstats_mock_scoreboard (Enable Mock Scoreboard Data) bool false

deathstats_afk_timeout (Scoreboard AFK Timeout Seconds) int 120 15 3600
